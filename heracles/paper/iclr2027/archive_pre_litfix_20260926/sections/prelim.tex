\section{Preliminaries}\label{sec:prelim}

We consider one-hidden-layer networks with homogeneous neurons and no biases, $f(\theta,x)=\sum_{i=1}^m\phi(\omega_i,x)$ with $\theta=\{\omega_1,\dots,\omega_m\}; \omega_i\in\sR^p$, where $\phi(\lambda\omega,x)=\lambda^\nu\phi(\omega,x)$ for all $\lambda>0$; $\nu$ is the \emph{homogeneity degree}. 

The $L_{2,b}$ norm is $\|\theta\|_{2,b}=\big(\sum_i\|\omega_i\|_2^b\big)^{1/b}$.

\paragraph{Modular addition} For a prime $p>2$ and $n\ge1$ operands, the dataset is $D_p:=\{((a_1,\dots,a_n),\sum_{1}^{k}a_k \mod p):a_k\in\sZ_p\}$; for an input $x$, $y(x)$ is its label, and $f(\theta,x)[c]$ is the logit of candidate $c\in\sZ_p$. Inputs are one-hot, so the embedding vector $u\in\R^p$ acts on operand $a$ as $u(a)$. The \emph{product neuron} is $\phi(\{u_1,\dots,u_n,w\},x)=w\prod_{1}^{n}u_n(a_n)$ and the \emph{power neuron} is $w\big(\sum_{1}^{n}u_n(a_n)\big)^n$ with $\nu=n+1$.

\paragraph{Fourier transform.} For $v:\sZ_p\to\sC$, $\hat v(j):=\sum_kv(k)\rho^{-jk}$ with $\rho:=e^{2\pi i/p}$. For real $v$, $\hat v(-j)=\overline{\hat v(j)}$, so the distinct non-zero frequencies are $\Fp:=\{1,\dots,\frac{p-1}2\}$. The multidimensional transform of the logit table is $\hat f_\theta(k_1,\dots,k_n;l):=\sum_{x,c}f(\theta,x)[c]\,\rho^{-(k_1a_1+\dots+k_na_n+lc)}$; for a product neuron it factorizes as $\hat w(l)\prod_{1}^{n}\hat u_n(k_n)$.
